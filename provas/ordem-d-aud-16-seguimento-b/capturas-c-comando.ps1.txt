# comando: o bloco abaixo, corrido pelo agente da sessão 1 (tmp/s1-agente.ps1, cópia do da D-AUD-16-SEGUIMENTO com a sede desta ordem; $sede = a pasta desta ordem no PC)
# ORDEM-D-AUD-16-SEGUIMENTO-B, corrida C (B-2): o cliente do Mestre com janela, servidor FORA DO AR, sem nunca ter estado na mesa.
Add-Type -Namespace W -Name S2 -MemberDefinition '[DllImport("user32.dll")] public static extern bool SetWindowPos(System.IntPtr h, System.IntPtr a, int x, int y, int cx, int cy, uint f);' -ErrorAction SilentlyContinue
$cliente = "$sede\repo\rascunhos\cliente"
$godot = '<godot 4.7.2>'
$antes = $env:APPDATA
$env:APPDATA = "$sede\tmp\appdata-c"
$t0 = Get-Date
"arranque: $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss zzz')"
$p = Start-Process -FilePath $godot -ArgumentList @('--path', "`"$cliente`"", '--windowed', '--resolution', '960x540', '--position', '0,40', '--', '--servidor=127.0.0.1:8790') -PassThru -WorkingDirectory $cliente
$env:APPDATA = $antes
"godot pid $($p.Id)"
while (((Get-Date)-$t0).TotalSeconds -lt 8) { Start-Sleep -Milliseconds 200 }
$script:M = JanelaDe $p.Id
[W.S2]::SetWindowPos($M, [IntPtr](-1), 0, 40, 976, 579, 0x40) | Out-Null
while (((Get-Date)-$t0).TotalSeconds -lt 13) { Start-Sleep -Milliseconds 200 }
"c01 (t=$([int]((Get-Date)-$t0).TotalSeconds) s): $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss zzz') $(Captura $M "$sede\tmp\cap\c01-mestre-servidor-fora-13s.png")"
while (((Get-Date)-$t0).TotalSeconds -lt 35) { Start-Sleep -Milliseconds 200 }
"c02 (t=$([int]((Get-Date)-$t0).TotalSeconds) s): $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss zzz') $(Captura $M "$sede\tmp\cap\c02-mestre-servidor-fora-35s.png")"
[W.S2]::SetWindowPos($M, [IntPtr](-2), 0, 0, 0, 0, 0x13) | Out-Null
Stop-Process -Id $p.Id -Force
"fechado: $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss zzz')"
